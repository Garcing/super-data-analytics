# visualizing-data Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a reusable `visualizing-data/` skill that renders deterministic single-chart PNG/SVG assets from structured JSON using Seaborn and Matplotlib.

**Architecture:** The skill exposes one stable CLI, `visualizing-data/scripts/chart.py`, backed by a small Python package under `scripts/chart_renderer/`. Shared modules own contract validation, IO, font setup, theme, dispatch, export, and quality checks; chart families live in focused renderer modules.

**Tech Stack:** Python 3, pandas, matplotlib, seaborn, Pillow for PNG quality checks, pytest for tests, existing system `skill-creator` scripts for skill metadata validation.

FIXTURES = ROOT / "tests" / "fixtures"


def run_chart(input_name: str, tmp_path: Path, *args: str) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        [sys.executable, str(CLI), str(FIXTURES / input_name), "--out-dir", str(tmp_path), *args],
        text=True,
        capture_output=True,
        check=False,
    )


def parse_stdout(result: subprocess.CompletedProcess[str]) -> dict:
    assert result.stdout.strip(), result.stderr
    return json.loads(result.stdout)


def image_changed_ratio(path: Path) -> float:
    with Image.open(path).convert("RGB") as image:
        diff = ImageChops.difference(image, Image.new("RGB", image.size, (255, 255, 255)))
        gray = diff.convert("L")
        changed = sum(count for value, count in enumerate(gray.histogram()) if value > 8)
        return changed / (image.width * image.height)

# visualizing-data/scripts/chart_renderer/io.py
from __future__ import annotations

import json
import re
from pathlib import Path
from typing import Any


class ChartInputError(ValueError):
    pass


def load_json(path: Path) -> dict[str, Any]:
    try:
        value = json.loads(path.read_text(encoding="utf-8"))
    except FileNotFoundError as exc:
        raise ChartInputError(f"input file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise ChartInputError(f"input file is not valid JSON: {exc.msg}") from exc
    if not isinstance(value, dict):
        raise ChartInputError("input JSON must be an object")
    return value


def slugify(value: str) -> str:
    cleaned = re.sub(r"[^0-9A-Za-z\u4e00-\u9fff_-]+", "-", value.strip()).strip("-")
    return cleaned[:80] or "chart"


def resolve_output_path(spec: dict[str, Any], *, out: str | None, out_dir: str | None, fmt: str) -> Path:
    if fmt not in {"png", "svg"}:
        raise ChartInputError("format must be png or svg")
    if out:
        path = Path(out)
        if path.suffix.lower() != f".{fmt}":
            path = path.with_suffix(f".{fmt}")
        path.parent.mkdir(parents=True, exist_ok=True)
        return path
    directory = Path(out_dir or ".")
    directory.mkdir(parents=True, exist_ok=True)
    title = str(spec.get("title", "chart"))
    return directory / f"{slugify(title)}.{fmt}"
```

# visualizing-data/scripts/chart_renderer/contract.py
from __future__ import annotations

from dataclasses import dataclass
from typing import Any


class ContractError(ValueError):
    pass


SUPPORTED_TYPES = {
    "area",
    "bar",
    "boxplot",
    "funnel",
    "heatmap",
    "histogram",
    "horizontal_bar",
    "line",
    "pareto",
    "pie",
    "scatter",
    "stacked_bar",
    "table",
    "waterfall",
}

NUMERIC_FIELDS = {
    "area": ["y"],
    "bar": ["y"],
    "boxplot": ["y"],
    "funnel": ["value"],
    "heatmap": ["value"],
    "histogram": ["x"],
    "horizontal_bar": ["y"],
    "line": ["y"],
    "pareto": ["y"],
    "pie": ["value"],
    "scatter": ["x", "y"],
    "stacked_bar": ["value"],
    "waterfall": ["delta"],
}


@dataclass(frozen=True)
class ChartSpec:
    chart_type: str
    title: str
    subtitle: str
    data: list[dict[str, Any]]
    encoding: dict[str, Any]
    options: dict[str, Any]


def require_text(spec: dict[str, Any], key: str) -> str:
    value = spec.get(key)
    if not isinstance(value, str) or not value.strip():
        raise ContractError(f"{key} is required and must be a non-empty string")
    return value.strip()


def require_rows(spec: dict[str, Any]) -> list[dict[str, Any]]:
    data = spec.get("data")
    if not isinstance(data, list):
        raise ContractError("data is required and must be a list of objects")
    rows = []
    for index, row in enumerate(data):
        if not isinstance(row, dict):
            raise ContractError(f"data[{index}] must be an object")
        rows.append(row)
    return rows


def field_name(encoding: dict[str, Any], role: str, *, required: bool = True) -> str | None:
    value = encoding.get(role)
    if value is None and not required:
        return None
    if not isinstance(value, str) or not value.strip():
        raise ContractError(f"encoding.{role} is required")
    return value.strip()


def require_field(rows: list[dict[str, Any]], field: str, role: str) -> None:
    if rows and all(field not in row for row in rows):
        raise ContractError(f"encoding.{role} field '{field}' is missing from data rows")


def ensure_numeric(rows: list[dict[str, Any]], field: str, role: str) -> None:
    values = [row.get(field) for row in rows if field in row and row.get(field) is not None]
    if values and not any(isinstance(value, (int, float)) and not isinstance(value, bool) for value in values):
        raise ContractError(f"encoding.{role} field '{field}' must contain numeric values")


def validate(spec: dict[str, Any]) -> ChartSpec:
    chart_type = spec.get("type")
    if chart_type not in SUPPORTED_TYPES:
        raise ContractError(f"type must be one of {sorted(SUPPORTED_TYPES)}")
    title = require_text(spec, "title")
    subtitle = require_text(spec, "subtitle")
    rows = require_rows(spec)
    encoding = spec.get("encoding")
    if not isinstance(encoding, dict):
        raise ContractError("encoding is required and must be an object")
    options = spec.get("options", {})
    if not isinstance(options, dict):
        raise ContractError("options must be an object when present")

    required_roles = {
        "area": ["x", "y"],
        "bar": ["x", "y"],
        "boxplot": ["x", "y"],
        "funnel": ["stage", "value"],
        "heatmap": ["x", "y", "value"],
        "histogram": ["x"],
        "horizontal_bar": ["x", "y"],
        "line": ["x", "y"],
        "pareto": ["x", "y"],
        "pie": ["label", "value"],
        "scatter": ["x", "y"],
        "stacked_bar": ["x", "series", "value"],
        "table": [],
        "waterfall": ["label", "delta"],
    }[chart_type]
    for role in required_roles:
        name = field_name(encoding, role)
        require_field(rows, name, role)
    for role in NUMERIC_FIELDS.get(chart_type, []):
        name = field_name(encoding, role)
        ensure_numeric(rows, name, role)
    return ChartSpec(chart_type, title, subtitle, rows, encoding, options)
```

from __future__ import annotations

from pathlib import Path


def ensure_output_exists(path: Path) -> None:
    if not path.exists() or path.stat().st_size == 0:
        raise RuntimeError(f"chart output was not created: {path}")


def ensure_png_nonblank(path: Path) -> None:
    if path.suffix.lower() != ".png":
        return
    from PIL import Image, ImageChops

    with Image.open(path).convert("RGB") as image:
        diff = ImageChops.difference(image, Image.new("RGB", image.size, (255, 255, 255)))
        if diff.getbbox() is None:
            raise RuntimeError(f"chart output appears blank: {path}")

# visualizing-data/scripts/chart_renderer/cli.py
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path
from typing import Any

from .contract import ContractError, validate
from .io import ChartInputError, load_json, resolve_output_path
from .quality import ensure_output_exists, ensure_png_nonblank
from .render import render_chart


def write_json(payload: dict[str, Any]) -> None:
    sys.stdout.write(json.dumps(payload, ensure_ascii=False) + "\n")


def parser() -> argparse.ArgumentParser:
    arg_parser = argparse.ArgumentParser(description="Render a chart image from JSON.")
    arg_parser.add_argument("input", help="Path to chart JSON input.")
    arg_parser.add_argument("--out", help="Output file path.")
    arg_parser.add_argument("--out-dir", help="Directory for generated chart output.")
    arg_parser.add_argument("--format", choices=["png", "svg"], default="png")
    return arg_parser


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    try:
        raw = load_json(Path(args.input))
        spec = validate(raw)
        output_path = resolve_output_path(raw, out=args.out, out_dir=args.out_dir, fmt=args.format)
        warnings = render_chart(spec, output_path, args.format)
        ensure_output_exists(output_path)
        ensure_png_nonblank(output_path)
        write_json({
            "ok": True,
            "path": str(output_path.resolve()),
            "format": args.format,
            "width": int(spec.options.get("width", 1200)),
            "height": int(spec.options.get("height", 720)),
            "warnings": warnings,
        })
        return 0
    except (ChartInputError, ContractError) as exc:
        write_json({"ok": False, "error": str(exc)})
        return 2
    except Exception as exc:
        write_json({"ok": False, "error": str(exc)})
        return 1
```

# visualizing-data/scripts/chart_renderer/fonts.py
from __future__ import annotations

import matplotlib
from matplotlib import font_manager

CJK_CANDIDATES = [
    "Microsoft YaHei",
    "SimHei",
    "Noto Sans CJK SC",
    "Source Han Sans SC",
    "PingFang SC",
    "WenQuanYi Micro Hei",
    "Arial Unicode MS",
]


def available_font_names() -> set[str]:
    return {font.name for font in font_manager.fontManager.ttflist}


def configure_fonts() -> list[str]:
    available = available_font_names()
    selected = [name for name in CJK_CANDIDATES if name in available]
    warnings: list[str] = []
    if not selected:
        warnings.append("No known CJK font detected; Chinese text may render as missing glyph boxes.")
    matplotlib.rcParams["font.sans-serif"] = [*selected, "DejaVu Sans", "Arial", "sans-serif"]
    matplotlib.rcParams["font.family"] = "sans-serif"
    matplotlib.rcParams["axes.unicode_minus"] = False
    return warnings

    "pink": {"xlight": "#FCDAD6", "light": "#F5BACC", "base": "#F390CA", "mid": "#BD569B", "dark": "#8A3A6F"},
}


def use_theme() -> None:
    sns.set_theme(
        style="whitegrid",
        rc={
            "figure.facecolor": TOKENS["surface"],
            "axes.facecolor": TOKENS["panel"],
            "axes.edgecolor": TOKENS["axis"],
            "axes.labelcolor": TOKENS["ink"],
            "xtick.color": TOKENS["muted"],
            "ytick.color": TOKENS["muted"],
            "grid.color": TOKENS["grid"],
            "grid.linewidth": 0.8,
            "axes.spines.top": False,
            "axes.spines.right": False,
            "savefig.facecolor": TOKENS["surface"],
            "savefig.edgecolor": "none",
        },
    )


def make_figure(width: int, height: int):
    return plt.subplots(figsize=(width / 150, height / 150), dpi=150)


def add_header(fig, ax, title: str, subtitle: str) -> None:
    title_text = textwrap.fill(title, width=72, break_long_words=False)
    subtitle_text = textwrap.fill(subtitle, width=100, break_long_words=False)
    ax.set_title("")
    fig.subplots_adjust(top=0.82)
    left = ax.get_position().x0
    fig.text(left, 0.965, title_text, ha="left", va="top", fontsize=13, fontweight="semibold", color=TOKENS["ink"])
    fig.text(left, 0.905, subtitle_text, ha="left", va="top", fontsize=9, color=TOKENS["muted"])
    sns.despine(ax=ax)


def palette(labels: Iterable[str], family_name: str = "blue") -> dict[str, str]:
    family = COLOR_FAMILIES[family_name]
    tones = ["base", "light", "xlight", "mid", "base"]
    return {label: family[tones[index % len(tones)]] for index, label in enumerate(labels)}

from __future__ import annotations

from importlib import import_module
from pathlib import Path

import matplotlib.pyplot as plt

from .contract import ChartSpec
from .fonts import configure_fonts
from .theme import use_theme

RENDERER_MODULES = {
    "area": "area",
    "bar": "bar",
    "horizontal_bar": "bar",
    "line": "line",
    "scatter": "scatter",
}


def render_chart(spec: ChartSpec, output_path: Path, fmt: str) -> list[str]:
    warnings = configure_fonts()
    use_theme()
    module_name = RENDERER_MODULES.get(spec.chart_type)
    if not module_name:
        raise RuntimeError(f"renderer for {spec.chart_type} is not implemented")
    module = import_module(f".renderers.{module_name}", package=__package__)
    fig = module.render(spec)
    try:
        fig.savefig(output_path, format=fmt, bbox_inches="tight")
    finally:
        plt.close(fig)
    return warnings

from __future__ import annotations

import numpy as np
import pandas as pd
from matplotlib.patches import Patch

from ..contract import ChartSpec
from ..theme import COLOR_FAMILIES, add_header, make_figure


def render(spec: ChartSpec):
    df = pd.DataFrame(spec.data)
    label = spec.encoding["label"]
    delta = spec.encoding["delta"]
    start = float(spec.options.get("start_value", 0))
    start_label = str(spec.options.get("start_label", "Start"))
    end_label = str(spec.options.get("end_label", "End"))
    running = start + df[delta].cumsum().shift(fill_value=0)
    ending = start + df[delta].cumsum()
    bottoms = np.minimum(running, ending)
    fig, ax = make_figure(int(spec.options.get("width", 1100)), int(spec.options.get("height", 640)))
    x_positions = np.arange(len(df) + 2)
    positive = COLOR_FAMILIES["olive"]
    negative = COLOR_FAMILIES["orange"]
    ax.bar(x_positions[0], start, color="#E2E5EA", edgecolor="#464C55", linewidth=1.0)
    for idx, row in df.iterrows():
        value = float(row[delta])
        family = positive if value >= 0 else negative
        ax.bar(x_positions[idx + 1], abs(value), bottom=bottoms.iloc[idx], color=family["base"], edgecolor=family["dark"], linewidth=1.0)
        ax.text(x_positions[idx + 1], ending.iloc[idx], f"{value:+g}", ha="center", va="bottom" if value >= 0 else "top", fontsize=8)
    end = float(ending.iloc[-1]) if len(ending) else start
    ax.bar(x_positions[-1], end, color="#E2E5EA", edgecolor="#464C55", linewidth=1.0)
    ax.set_xticks(x_positions, [start_label, *df[label].astype(str).tolist(), end_label], rotation=0)
    ax.legend(handles=[
        Patch(facecolor=positive["base"], edgecolor=positive["dark"], label="Positive"),
        Patch(facecolor=negative["base"], edgecolor=negative["dark"], label="Negative"),
        Patch(facecolor="#E2E5EA", edgecolor="#464C55", label="Start/end"),
    ], loc="lower left", bbox_to_anchor=(0, 1.02), frameon=False, ncol=3)
    add_header(fig, ax, spec.title, spec.subtitle)
    return fig

# Visualizing Data

Generate standalone chart images from structured JSON using the bundled Seaborn/Matplotlib renderer.

## Default Workflow

1. Read `references/input_contract.md` before preparing JSON by hand.
2. Create a chart JSON file with `type`, `title`, `subtitle`, `data`, `encoding`, and optional `options`.
3. Run `python visualizing-data/scripts/chart.py <input.json>` for PNG.
4. Use `--format svg` when the downstream workflow needs SVG.
5. Check the CLI JSON result. Do not claim success if `ok` is false or the output file is missing.

## Use This Instead Of AI Image Generation When

- The chart contains exact values, axis labels, or metric labels.
- The output must be reproducible.
- A report, document, slide, PDF, or message needs a chart asset rather than a full generated poster.

## Chinese Text

Chinese titles, subtitles, legends, labels, value labels, and table cells are supported. The renderer configures a CJK font fallback chain and disables Matplotlib Unicode-minus rendering issues. If the CLI returns a CJK font warning, mention that the chart was generated but the local machine may need a Chinese font such as Microsoft YaHei, SimHei, Noto Sans CJK SC, or Source Han Sans SC for best results.

## References

- `references/input_contract.md`: accepted JSON fields and chart-specific examples.
- `references/seaborn_templates.md`: visual style guidance for changing renderers or adding chart types.

# visualizing-data Input Contract

The renderer accepts one JSON object.

## Common Fields

| Field | Required | Meaning |
|---|---:|---|
| `type` | yes | One of `bar`, `horizontal_bar`, `line`, `area`, `pie`, `scatter`, `heatmap`, `histogram`, `boxplot`, `stacked_bar`, `waterfall`, `funnel`, `pareto`, `table` |
| `title` | yes | Visible chart title |
| `subtitle` | yes | Visible metric scope, time window, unit, denominator, or note |
| `data` | yes | Array of row objects |
| `encoding` | yes | Field mapping for the selected chart type |
| `options` | no | Width, height, sorting, labels, bins, and chart-specific settings |

## CLI

```bash
python visualizing-data/scripts/chart.py input.json
python visualizing-data/scripts/chart.py input.json --format svg --out output/chart.svg
```

## Chart Encodings

| Type | Required encoding |
|---|---|
| `bar`, `horizontal_bar`, `line`, `area`, `boxplot` | `x`, `y` |
| `scatter` | `x`, `y` |
| `histogram` | `x` |
| `pie` | `label`, `value` |
| `heatmap` | `x`, `y`, `value` |
| `stacked_bar` | `x`, `series`, `value` |
| `waterfall` | `label`, `delta`; options include `start_value`, `start_label`, `end_label` |
| `funnel` | `stage`, `value` |
| `pareto` | `x`, `y` |
| `table` | no required encoding |

## Chinese Example

```json
{
  "type": "bar",
  "title": "各区域销售额",
  "subtitle": "2026年6月，单位：万元",
  "data": [
    { "region": "华东", "sales": 120 },
    { "region": "华北", "sales": -35 }
  ],
  "encoding": { "x": "region", "y": "sales" },
  "options": { "value_labels": "auto", "width": 900, "height": 560 }
}
```
